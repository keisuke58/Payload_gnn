\textbf{Defects:} debonding / FOD / impact / delamination

\begin{block}{(A) Static FEM analysis}

  \item \textbf{10{,}897 nodes} $\times$ \textbf{34 features} (disp, stress, temp, layup, \ldots)

  \item Model: MIFNO-GW (2-D factorised FNO, waveform-only)

\section{Methods}

\begin{frame}{Three modelling additions}
\begin{block}{(A) Multi-task learning — defect centroid regression}
Joint loss: $\mathcal{L} = \mathcal{L}_\text{focal} + \lambda_\text{reg}\,\mathcal{L}_\text{reg}$\\
Shared backbone $\to$ node classification head + graph-level MLP for $(x,y,z)$ centroid.\\
GT centroid computed on-the-fly from defective nodes.
\end{block}
\vspace{4pt}
\begin{block}{(B) Level-1 PINN — stress gradient regularisation}
High predicted defect probability should co-locate with high $|\nabla\sigma_\text{Mises}|$.\\
$\mathcal{L}_\text{stress} = \frac{1}{N}\sum_i p_i^{(\text{def})}\,\exp(-|\Delta\sigma_i|/\sigma_0)$
\end{block}
\vspace{4pt}
\begin{block}{(C) Level-2 PINN — static equilibrium residual $\nabla\!\cdot\!\boldsymbol{\sigma}=0$}
Graph-edge finite differences on $(s_{11},s_{22},s_{12})$; penalises defect predictions
at nodes with low equilibrium violation.
\end{block}
\end{frame}

\begin{frame}{LGSTA — Local-Global Attention GNN}
\textbf{Best-performing architecture:} LocalGlobalAttentionGNN (inspired by LGSTA-GNN, Buildings 2025)
\vspace{0.3em}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Local branch} (GAT $\times$ 4 layers)
\begin{itemize}
  \item Neighbourhood message-passing
  \item Captures local defect boundaries and stress concentrations
\end{itemize}
\vspace{6pt}
\textbf{Global branch} (Multi-head self-attention)
\begin{itemize}
  \item Full $N \times N$ attention over all 10{,}897 nodes
  \item Captures global mode shapes and structural coupling
\end{itemize}
\vspace{6pt}
\textbf{Fusion:} gated combination of both representations
\end{column}
\begin{column}{0.48\textwidth}
\centering

\begin{column}{0.48\textwidth}
\begin{block}{Data quality fix (v1 → v2)}
\textbf{Root cause found:} healthy CSVs (9 sensors, circumferential ring)
and defect CSVs (99 sensors, axial line) used \emph{different} physical
sensor layouts. v1 F1\,=\,1.0 was a position-leakage artifact.

\smallskip
\textbf{Fix:} filter to 9-sensor position-consistent subset (108 samples);
remove position branch (waveform-only FNO). New F1\,=\,1.0 is \emph{genuine}
— confirmed on held-out validation split.
\end{block}
\vspace{4pt}
\begin{block}{Next: v3 with 197-sample balanced dataset}
Abaqus Explicit (99-sensor healthy job) running on frontale04.
Upon completion: 197 position-consistent samples (81 healthy + 116 defect)
$\Rightarrow$ enables robust L3-PINN training (dispersion loss).
\end{block}

% ---- key findings -----------------------------------------------
\begin{frame}{Key findings}
\begin{block}{(A-1) LGSTA wins at OptF1 = \best{0.896} (interim ep\,58)}
Dual local-global attention: local GAT finds defect boundaries; global self-attention captures full structural coupling.
Node-cls only (NC) outperforms multitask (MT, 0.846): centroid regression branch may compete for capacity.
\end{block}
\vspace{3pt}
\begin{block}{(A-2) Multitask centroid regression: 0.7--0.9\,mm RMSE at sub-200 epochs}
Shared backbone simultaneously classifies defect type \emph{and} locates centroid — sub-mm accuracy on a 5\,m structure.
\end{block}
\vspace{3pt}
\begin{block}{(B) MIFNO-GW v2: F1\,=\,1.000, AUC\,=\,1.000 (waveform-only FNO)}
Position-leakage artifact eliminated. v2\_nodip ($\lambda=0$) and v2\_pinn ($\lambda=0.05$) both converge to perfect validation performance.
Small dataset caveat: 5 defect train samples; v3 will use 197-sample balanced set.
\end{block}
\vspace{3pt}

\begin{column}{0.48\textwidth}
\begin{alertblock}{(A-3) Transolver collapses (AUC\,$\approx$\,0.49)}
Designed for structured grids; fails on irregular FEM mesh.
